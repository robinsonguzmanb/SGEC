import subprocess
import sys
import os

def main():
    print("=== Iniciando Constructor del Instalador SGEC ===")
    
    # 1. Asegurar que PyInstaller esta instalado
    try:
        import PyInstaller
    except ImportError:
        print("Instalando PyInstaller...")
        subprocess.run([sys.executable, "-m", "pip", "install", "pyinstaller"], check=True)
    
    # 2. Comando para empaquetar
    print("Compilando el ejecutable SGEC_Instalador.exe...")
    
    # AÃ±adimos los recursos (archivos y carpetas) usando --add-data
    # Sintaxis Windows: 'origen;destino', Linux/Mac: 'origen:destino'
    # Como el script se puede ejecutar en diferentes OS para generar el paquete,
    # detectamos el separador adecuado
    sep = ';' if os.name == 'nt' else ':'
    
    cmd = [
        sys.executable, "-m", "PyInstaller",
        "--onefile",
        "--name", "SGEC_Instalador",
        "--uac-admin",  # Pide permisos de administrador en Windows al ejecutarse
        f"--add-data=app.py{sep}.",
        f"--add-data=requirements.txt{sep}.",
        f"--add-data=db{sep}db",
        f"--add-data=pages{sep}pages",
        f"--add-data=utils{sep}utils",
        "launcher.py"
    ]
    
    # Ejecutar PyInstaller
    subprocess.run(cmd, check=True)
    
    print("\n========================================================")
    print("CONSTRUCCION FINALIZADA.")
    print("El archivo ejecutable se encuentra en: dist/SGEC_Instalador.exe")
    print("========================================================")

if __name__ == "__main__":
    main()

import os

import shutil
import subprocess
import urllib.request

def resource_path(relative_path):
    """ Get absolute path to resource, works for dev and for PyInstaller """
    try:
        # PyInstaller creates a temp folder and stores path in _MEIPASS
        base_path = sys._MEIPASS
    except Exception:
        base_path = os.path.abspath(".")
    return os.path.join(base_path, relative_path)

def main():
    INSTALL_DIR = r"C:\SGEC"
    try:
        print("Iniciando instalacion de SGEC en " + INSTALL_DIR)

        # 1. Extraer archivos a C:\SGEC
        if not os.path.exists(INSTALL_DIR):
            os.makedirs(INSTALL_DIR, exist_ok=True)
            
        for item in ['app.py', 'requirements.txt', 'db', 'pages', 'utils']:
            src = resource_path(item)
            dst = os.path.join(INSTALL_DIR, item)
            if os.path.exists(src):
                if os.path.isdir(src):
                    shutil.copytree(src, dst, dirs_exist_ok=True)
                else:
                    shutil.copy2(src, dst)
                    
        # Crear carpetas outputs y db si no existen
        os.makedirs(os.path.join(INSTALL_DIR, "outputs", "cartas"), exist_ok=True)
        os.makedirs(os.path.join(INSTALL_DIR, "db"), exist_ok=True)

        # 2. Verificar e instalar Ollama
        print("Verificando Ollama...")
        try:
            subprocess.run(["where", "ollama"], check=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        except subprocess.CalledProcessError:
            print("Ollama no fue detectado. Iniciando instalacion...")
            ollama_path = os.path.join(INSTALL_DIR, "OllamaSetup.exe")
            
            def show_progress(block_num, block_size, total_size):
                downloaded = block_num * block_size
                if total_size > 0:
                    percent = downloaded * 100 / total_size
                    percent = min(percent, 100) # Cap at 100%
                    bar_length = 40
                    filled_length = int(bar_length * percent // 100)
                    bar = 'â–ˆ' * filled_length + '-' * (bar_length - filled_length)
                    sys.stdout.write(f'\rDescargando OllamaSetup.exe: |{bar}| {percent:.1f}% ({downloaded/(1024*1024):.1f} MB / {total_size/(1024*1024):.1f} MB)')
                    sys.stdout.flush()
                else:
                    sys.stdout.write(f'\rDescargando OllamaSetup.exe: {downloaded/(1024*1024):.1f} MB')
                    sys.stdout.flush()

            print("Descargando el instalador oficial de Ollama...")
            urllib.request.urlretrieve('https://ollama.com/download/OllamaSetup.exe', ollama_path, reporthook=show_progress)
            print("\nEjecutando instalador de Ollama. Por favor, sigue las instrucciones de la ventana emergente...")
            subprocess.run([ollama_path], check=False)
            try:
                os.remove(ollama_path)
            except Exception:

        print("\n---------------------------------------------------------")
        print("Iniciando descarga del cerebro de IA (modelo Llama 3).")
        print("Tamano aproximado: 4.7 GB. El tiempo dependera de tu conexion a internet.")
        print("Por favor, NO cierres esta ventana. El proceso continua...")
        print("---------------------------------------------------------")
        
        # Run interactive so the user sees the progress bar provided natively by ollama
        subprocess.run(["ollama", "pull", "llama3"], check=False)

        # 3. Configurar Entorno Python
        venv_dir = os.path.join(INSTALL_DIR, "venv")
        if not os.path.exists(venv_dir):
            print("Creando entorno virtual...")
            # NOTA IMPORTANTE: No usar sys.executable aquÃ­ cuando se corre desde PyInstaller,

            subprocess.run(["python", "-m", "venv", venv_dir], check=True)

from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2HMAC
import base64

DB_PATH = 'sgec.db'

def get_fernet(secret_key):
    """Deriva una llave segura Fernet de 32 bytes a partir del texto ingresado por el Admin."""
    salt = b'sgec_salt_v1'
    kdf = PBKDF2HMAC(
        algorithm=hashes.SHA256(),
        length=32,
        salt=salt,
        iterations=100000,
    )
    key = base64.urlsafe_b64encode(kdf.derive(secret_key.encode()))
    return Fernet(key)

def encrypt_val(fernet_obj, val):
    if val is None or pd.isna(val) or str(val).strip() == "":
        return None
    return fernet_obj.encrypt(str(val).encode()).decode()

def decrypt_val(fernet_obj, val):
    if val is None or str(val).strip() == "":
        return None
    try:
        return fernet_obj.decrypt(val.encode()).decode()
    except Exception:
        return val # En caso de que se pase un dato en claro sin querer

    """Obtiene la conexiÃ³n a la base de datos SQLite con WAL mode."""

    conn.execute("PRAGMA synchronous=NORMAL;")
    return conn

def execute_query(query, params=()):
    """Ejecuta una consulta SQL genÃ©rica que retorna resultados."""
    conn = get_connection()
    try:
        cursor = conn.cursor()
        cursor.execute(query, params)

def save_dataframe_to_table(secret_key, df, table_name):
    """Guarda un DataFrame. Si es Empleados, encripta los datos en vivo."""
    conn = get_connection()
    try:
        if table_name == 'Empleados':
            f = get_fernet(secret_key)
            df_enc = df.copy()
            for col in ['nombre_completo', 'grado_actual', 'sueldo_base']:
                if col in df_enc.columns:
                    df_enc[col] = df_enc[col].apply(lambda x: encrypt_val(f, x))
            df_enc.to_sql(table_name, conn, if_exists='append', index=False)
        else:
            df.to_sql(table_name, conn, if_exists='append', index=False)
        conn.commit()
    finally:
        conn.close()

def _decrypt_empleado_row(f, r):
    r['nombre_completo'] = decrypt_val(f, r['nombre_completo'])
    r['grado_actual'] = decrypt_val(f, r['grado_actual'])
    try:
        r['sueldo_base'] = float(decrypt_val(f, r['sueldo_base']))
    except:
        r['sueldo_base'] = 0.0
    return r

            df_reservas.to_excel(writer, sheet_name='Embudo y Reservas', index=False)

    return output.getvalue()
